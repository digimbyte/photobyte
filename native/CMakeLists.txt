cmake_minimum_required(VERSION 3.20)
project(DigiPhotoCodecs LANGUAGES C)

set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DIGI_ZLIB}" zlib EXCLUDE_FROM_ALL)

configure_file("${DIGI_PNG}/scripts/pnglibconf.h.prebuilt" "${CMAKE_CURRENT_BINARY_DIR}/pnglibconf.h" COPYONLY)
set(png_names png pngerror pngget pngmem pngpread pngread pngrio pngrtran pngrutil pngset pngtrans pngwio pngwrite pngwtran pngwutil)
foreach(name IN LISTS png_names)
    list(APPEND png_sources "${DIGI_PNG}/${name}.c")
endforeach()
add_library(digipng STATIC ${png_sources})
target_include_directories(digipng PUBLIC "${DIGI_PNG}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(digipng PUBLIC zlibstatic)

file(GLOB lcms_sources CONFIGURE_DEPENDS "${DIGI_LCMS}/src/*.c")
add_library(digicms STATIC ${lcms_sources})
target_include_directories(digicms PUBLIC "${DIGI_LCMS}/include")
target_compile_definitions(digicms PRIVATE _CRT_SECURE_NO_WARNINGS)

foreach(tool ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
    set(WEBP_BUILD_${tool} OFF CACHE BOOL "" FORCE)
endforeach()
set(WEBP_USE_THREAD OFF CACHE BOOL "" FORCE)
add_subdirectory("${DIGI_WEBP}" webp EXCLUDE_FROM_ALL)

add_library(digiphoto_codecs SHARED bridge.c)
target_include_directories(digiphoto_codecs PRIVATE "${DIGI_WEBP}/src")
target_link_libraries(digiphoto_codecs PRIVATE digipng digicms webp)
